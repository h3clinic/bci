[J5_CARRIER]
type = "Samtec QSH-030-01-L-D-A"

mating_height_mm = 5
mating_connector = "QTH-030-01-L-D-A (carrier side)"

# ─────────────────────────────────────────────────────────────────────────
# PINOUT — Ground-Signal-Ground (GSG) pattern for high-speed integrity
# ─────────────────────────────────────────────────────────────────────────
#
# Row A (odd pins)        Row B (even pins)
# ─────────────          ─────────────
#  1: GND                  2: GND             ← Ground fence top
#  3: SCLK                 4: GND             ← Clock + adjacent ground
#  5: MOSI                 6: GND             ← Data + adjacent ground
#  7: CS1                  8: CS2             ← Chip selects (grouped)
#  9: GND                 10: GND             ← Ground fence mid
# 11: MISO1_A             12: MISO1_B         ← U1 DDR MISO pair
# 13: GND                 14: GND             ← Ground between MISO groups
# 15: MISO2_A             16: MISO2_B         ← U2 DDR MISO pair
# 17: GND                 18: GND             ← Ground fence
# 19: VIN                 20: VIN             ← Power (paralleled)
# 21: GND                 22: GND             ← Power return
# 23: VIN                 24: VIN             ← Power (paralleled, 4 pins total)
# 25: GND                 26: GND             ← Power return
# 27: TEST_SHORT_EN       28: CAL_EN          ← Test control signals
# 29: SPARE               30: GND             ← Spare + ground

# Pin assignments (canonical)
pin_1  = GND
pin_2  = GND
pin_3  = SCLK              # Carrier → Headstage | 20 MHz | 3.3V LVCMOS
pin_4  = GND               # Return for SCLK
pin_5  = MOSI              # Carrier → Headstage | 20 Mbps | 3.3V LVCMOS
pin_6  = GND               # Return for MOSI
pin_7  = CS1               # Carrier → Headstage | Active low | U1 select
pin_8  = CS2               # Carrier → Headstage | Active low | U2 select
pin_9  = GND
pin_10 = GND
pin_11 = MISO1_A           # Headstage → Carrier | U1 ch0-31 | DDR | 3.3V
pin_12 = MISO1_B           # Headstage → Carrier | U1 ch32-63 | DDR | 3.3V
pin_13 = GND
pin_14 = GND
pin_15 = MISO2_A           # Headstage → Carrier | U2 ch0-31 | DDR | 3.3V
pin_16 = MISO2_B           # Headstage → Carrier | U2 ch32-63 | DDR | 3.3V
pin_17 = GND
pin_18 = GND
pin_19 = VIN               # Carrier → Headstage | 3.3–5V power
pin_20 = VIN               # Carrier → Headstage | paralleled
pin_21 = GND               # Power return
pin_22 = GND               # Power return
pin_23 = VIN               # Carrier → Headstage | paralleled
pin_24 = VIN               # Carrier → Headstage | paralleled
pin_25 = GND               # Power return
pin_26 = GND               # Power return
pin_27 = TEST_SHORT_EN     # Carrier → Headstage | Input shorting control
pin_28 = CAL_EN            # Carrier → Headstage | Cal injection enable
pin_29 = SPARE             # Reserved for future use (e.g. RESET, AUX)
pin_30 = GND

# ─────────────────────────────────────────────────────────────────────────
# SIGNAL SUMMARY
# ─────────────────────────────────────────────────────────────────────────
# Direction         Signal         Count   Pins
# ────────────────────────────────────────────────────
# Carrier→Headstage SCLK           1       3
# Carrier→Headstage MOSI           1       5
# Carrier→Headstage CS1            1       7
# Carrier→Headstage CS2            1       8
# Headstage→Carrier MISO1_A        1       11
# Headstage→Carrier MISO1_B        1       12
# Headstage→Carrier MISO2_A        1       15
# Headstage→Carrier MISO2_B        1       16
# Carrier→Headstage TEST_SHORT_EN  1       27
# Carrier→Headstage CAL_EN         1       28
# Reserved          SPARE          1       29
# Power             VIN            4       19,20,23,24
# Ground            GND            14      1,2,4,6,9,10,13,14,17,18,21,22,25,26,30
# ────────────────────────────────────────────────────
# Total: 30 pins (10 signal + 4 power + 1 spare + 14 ground + 1 ground)
# Ground ratio: 15/30 = 50% (meets GSG target)

# ELECTRICAL RULES:
# • Every high-speed signal has at least one adjacent ground pin
# • SCLK (pin 3) flanked by GND pins 1,2 and 4
# • MOSI (pin 5) flanked by GND pins 4 and 6
# • MISO pairs (11/12, 15/16) flanked by GND 9/10 and 13/14 and 17/18
# • VIN uses 4 pins (rated 0.5A each → 2A capacity, need <200mA)
# • Spare pin allows future RESET, AUX_ADC, or debug signal
